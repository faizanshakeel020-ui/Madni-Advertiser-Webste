     5→     5→ * Add/edit/delete clients (round logo) and their projects (shown on logo click).
     6→     6→ */
     7→     7→import { useEffect, useRef, useState } from "react";
     8→     8→import {
     9→     9→  Briefcase,
    10→    10→  Images,
    11→    11→  Loader2,
    12→    12→  Pencil,
    13→    13→  Plus,
    14→    14→  Sparkles,
    15→    15→  Trash2,
    16→    16→  Upload,
    17→    17→  Wand2,
    18→    18→  X,
    19→    19→} from "lucide-react";
    20→    20→import { toast } from "sonner";
    21→    21→import { Button } from "@/components/ui/button";
    22→    22→import { Input } from "@/components/ui/input";
    23→    23→import { Label } from "@/components/ui/label";
    24→    24→import { Textarea } from "@/components/ui/textarea";
    25→    25→import { Badge } from "@/components/ui/badge";
    26→    26→import { Skeleton } from "@/components/ui/skeleton";
    27→    27→import {
    28→    28→  Dialog,
    29→    29→  DialogContent,
    30→    30→  DialogDescription,
    31→    31→  DialogHeader,
    32→    32→  DialogTitle,
    33→    33→} from "@/components/ui/dialog";
    34→    34→import {
    35→    35→  AlertDialog,
    36→    36→  AlertDialogAction,
    37→    37→  AlertDialogCancel,
    38→    38→  AlertDialogContent,
    39→    39→  AlertDialogDescription,
    40→    40→  AlertDialogFooter,
    41→    41→  AlertDialogHeader,
    42→    42→  AlertDialogTitle,
    43→    43→} from "@/components/ui/alert-dialog";
    44→    44→import {
    45→    45→  adminDeleteClient,
    46→    46→  adminDeleteClientProject,
    47→    47→  adminFetchClients,
    48→    48→  adminGenerateProjectDescription,
    49→    49→  adminSaveClient,
    50→    50→  adminSaveClientProject,
    51→    51→  uploadImage,
    52→    52→  uploadImages,
    53→    53→} from "@/lib/api";
    54→    54→import type { GenerateDescriptionResponse } from "@/lib/api";
    55→    55→import { slugify } from "@/lib/format";
    56→    56→import type { Client, ClientProject } from "@/lib/types";
    57→    57→
    58→    58→type ClientEdit = {
    59→    59→  id?: string;
    60→    60→  name: string;
    61→    61→  slug: string;
    62→    62→  logo: string;
    63→    63→  industry: string;
    64→    64→};
    65→    65→
    66→    66→type ProjectEdit = {
    67→    67→  id?: string;
    68→    68→  clientId: string;
    69→    69→  clientName: string;
    70→    70→  title: string;
    71→    71→  description: string;
    72→    72→  images: string[];
    73→    73→  year: string;
    74→    74→};
    75→    75→
    76→    76→const emptyClient: ClientEdit = { name: "", slug: "", logo: "", industry: "" };
    77→    77→
    78→    78→export function AdminClients() {
    79→    79→  const [clients, setClients] = useState<Client[]>([]);
    80→    80→  const [loading, setLoading] = useState(true);
    81→    81→  const [saving, setSaving] = useState(false);
    82→    82→
    83→    83→  // client dialog
    84→    84→  const [clientOpen, setClientOpen] = useState(false);
    85→    85→  const [clientEdit, setClientEdit] = useState<ClientEdit | null>(null);
    86→    86→  const [uploadingLogo, setUploadingLogo] = useState(false);
    87→    87→  const [deleteClient, setDeleteClient] = useState<Client | null>(null);
    88→    88→
    89→    89→  // project dialog
    90→    90→  const [projectOpen, setProjectOpen] = useState(false);
    91→    91→  const [projectEdit, setProjectEdit] = useState<ProjectEdit | null>(null);
    92→    92→  const [uploadingProjImg, setUploadingProjImg] = useState(false);
    93→    93→  const [projImgUrlInput, setProjImgUrlInput] = useState("");
    94→    94→  const [deleteProject, setDeleteProject] = useState<ClientProject | null>(null);
    95→    95→
    96→    96→  // project description AI writer
    97→    97→  const [generatingDesc, setGeneratingDesc] = useState(false);
    98→    98→  const [projSeoInfo, setProjSeoInfo] = useState<GenerateDescriptionResponse | null>(null);
    99→    99→  const projDescTouched = useRef(false);
   100→   100→  const projLastGenKey = useRef("");
   101→   101→  const projVariationRef = useRef(1);
   102→   102→
   103→   103→  const load = async () => {
   104→   104→    setLoading(true);
   105→   105→    try {
   106→   106→      setClients(await adminFetchClients());
   107→   107→    } catch {
   108→   108→      toast.error("Failed to load clients");
   109→   109→    } finally {
   110→   110→      setLoading(false);
   111→   111→    }
   112→   112→  };
   113→   113→
   114→   114→  useEffect(() => {
   115→   115→    load();
   116→   116→  }, []);
   117→   117→
   118→   118→  // ---------- client CRUD ----------
   119→   119→
   120→   120→  const openNewClient = () => {
   121→   121→    setClientEdit({ ...emptyClient });
   122→   122→    setClientOpen(true);
   123→   123→  };
   124→   124→
   125→   125→  const openEditClient = (c: Client) => {
   126→   126→    setClientEdit({ id: c.id, name: c.name, slug: c.slug, logo: c.logo, industry: c.industry ?? "" });
   127→   127→    setClientOpen(true);
   128→   128→  };
   129→   129→
   130→   130→  const saveClient = async () => {
   131→   131→    if (!clientEdit) return;
   132→   132→    if (!clientEdit.name.trim()) return toast.error("Client name is required");
   133→   133→    if (!clientEdit.logo.trim()) return toast.error("Upload or paste a client logo image");
   134→   134→    setSaving(true);
   135→   135→    try {
   136→   136→      await adminSaveClient({
   137→   137→        id: clientEdit.id,
   138→   138→        name: clientEdit.name.trim(),
   139→   139→        slug: clientEdit.slug.trim(),
   140→   140→        logo: clientEdit.logo.trim(),
   141→   141→        industry: clientEdit.industry.trim() || null,
   142→   142→      });
   143→   143→      toast.success(clientEdit.id ? "Client updated" : "Client added");
   144→   144→      setClientOpen(false);
   145→   145→      await load();
   146→   146→    } catch (e) {
   147→   147→      toast.error("Save failed", { description: e instanceof Error ? e.message : "" });
   148→   148→    } finally {
   149→   149→      setSaving(false);
   150→   150→    }
   151→   151→  };
   152→   152→
   153→   153→  const handleLogoUpload = async (file: File) => {
   154→   154→    setUploadingLogo(true);
   155→   155→    try {
   156→   156→      const { url } = await uploadImage(file);
   157→   157→      setClientEdit((s) => (s ? { ...s, logo: url } : s));
   158→   158→      toast.success("Logo uploaded");
   159→   159→    } catch (e) {
   160→   160→      toast.error("Upload failed", { description: e instanceof Error ? e.message : "" });
   161→   161→    } finally {
   162→   162→      setUploadingLogo(false);
   163→   163→    }
   164→   164→  };
   165→   165→
   166→   166→  // ---------- project CRUD ----------
   167→   167→
   168→   168→  /** AI-write a long SEO case-study description and fill the textarea. */
   169→   169→  const runGenerateProjDesc = async (src: ProjectEdit, variation: number) => {
   170→   170→    setGeneratingDesc(true);
   171→   171→    try {
   172→   172→      // short existing text = the admin's notes → use as hints for the writer
   173→   173→      const notes = src.description.trim();
   174→   174→      const hints = notes.length > 0 && notes.length < 120 ? notes : undefined;
   175→   175→      const res = await adminGenerateProjectDescription({
   176→   176→        clientId: src.clientId,
   177→   177→        clientName: src.clientName,
   178→   178→        title: src.title.trim(),
   179→   179→        year: src.year ? Number(src.year) : null,
   180→   180→        hints,
   181→   181→        variation,
   182→   182→      });
   183→   183→      setProjectEdit((s) => (s ? { ...s, description: res.description } : s));
   184→   184→      setProjSeoInfo(res);
   185→   185→      if (res.method === "ai") {
   186→   186→        toast.success("AI wrote a long SEO case study", {
   187→   187→          description: `${res.wordCount} words · ${res.keywords.length} ranking keywords built in. Edit freely.`,
   188→   188→        });
   189→   189→      } else {
   190→   190→        toast.info("Description generated", {
   191→   191→          description: "AI was busy — an SEO template was used. Try Regenerate for a fresh AI draft.",
   192→   192→        });
   193→   193→      }
   194→   194→    } catch {
   195→   195→      toast.error("AI description failed", { description: "Write it manually or try again." });
   196→   196→    } finally {
   197→   197→      setGeneratingDesc(false);
   198→   198→    }
   199→   199→  };
   200→   200→
   201→   201→  const openNewProject = (c: Client) => {
   202→   202→    projDescTouched.current = false;
   203→   203→    projLastGenKey.current = "";
   204→   204→    projVariationRef.current = 1;
   205→   205→    setProjSeoInfo(null);
   206→   206→    setProjImgUrlInput("");
   207→   207→    setProjectEdit({
   208→   208→      clientId: c.id,
   209→   209→      clientName: c.name,
   210→   210→      title: "",
   211→   211→      description: "",
   212→   212→      images: [],
   213→   213→      year: String(new Date().getFullYear()),
   214→   214→    });
   215→   215→    setProjectOpen(true);
   216→   216→  };
   217→   217→
   218→   218→  const openEditProject = (c: Client, p: ClientProject) => {
   219→   219→    projDescTouched.current = true; // never auto-overwrite an existing story
   220→   220→    projLastGenKey.current = p.title;
   221→   221→    projVariationRef.current = 1;
   222→   222→    setProjSeoInfo(null);
   223→   223→    setProjImgUrlInput("");
   224→   224→    setProjectEdit({
   225→   225→      id: p.id,
   226→   226→      clientId: c.id,
   227→   227→      clientName: c.name,
   228→   228→      title: p.title,
   229→   229→      description: p.description,
   230→   230→      images: [...p.images],
   231→   231→      year: p.year ? String(p.year) : "",
   232→   232→    });
   233→   233→    setProjectOpen(true);
   234→   234→  };
   235→   235→
   236→   236→  const generateProjDescNow = () => {
   237→   237→    if (!projectEdit || !projectEdit.title.trim()) {
   238→   238→      return toast.error("Type a project title first");
   239→   239→    }
   240→   240→    projVariationRef.current += 1;
   241→   241→    projDescTouched.current = false; // fresh AI text may replace what's there
   242→   242→    projLastGenKey.current = projectEdit.title.trim();
   243→   243→    setProjSeoInfo(null);
   244→   244→    void runGenerateProjDesc(projectEdit, projVariationRef.current);
   245→   245→  };
   246→   246→
   247→   247→  // Auto-write the long description for a NEW project once the title settles
   248→   248→  // (debounced 2.5s, only while empty, never after manual typing or a failed attempt)
   249→   249→  useEffect(() => {
   250→   250→    if (!projectEdit || projectEdit.id || projDescTouched.current || generatingDesc) return;
   251→   251→    const title = projectEdit.title.trim();
   252→   252→    if (title.length < 6 || projectEdit.description.trim()) return;
   253→   253→    if (title === projLastGenKey.current) return;
   254→   254→    const t = setTimeout(() => {
   255→   255→      projLastGenKey.current = title;
   256→   256→      void runGenerateProjDesc(projectEdit, 1);
   257→   257→    }, 2500);
   258→   258→    return () => clearTimeout(t);
   259→   259→  }, [projectEdit, generatingDesc]);
   260→   260→
   261→   261→  const saveProject = async () => {
   262→   262→    if (!projectEdit) return;
   263→   263→    if (!projectEdit.title.trim()) return toast.error("Project title is required");
   264→   264→    if (!projectEdit.description.trim()) return toast.error("Project description is required");
   265→   265→    if (projectEdit.images.length === 0) return toast.error("Upload at least one project image");
   266→   266→    setSaving(true);
   267→   267→    try {
   268→   268→      await adminSaveClientProject({
   269→   269→        id: projectEdit.id,
   270→   270→        clientId: projectEdit.clientId,
   271→   271→        title: projectEdit.title.trim(),
   272→   272→        description: projectEdit.description.trim(),
   273→   273→        images: projectEdit.images,
   274→   274→        year: projectEdit.year ? Number(projectEdit.year) : null,
   275→   275→      });
   276→   276→      toast.success(projectEdit.id ? "Project updated" : "Project added");
   277→   277→      setProjectOpen(false);
   278→   278→      await load();
   279→   279→    } catch (e) {
   280→   280→      toast.error("Save failed", { description: e instanceof Error ? e.message : "" });
   281→   281→    } finally {
   282→   282→      setSaving(false);
   283→   283→    }
   284→   284→  };
   285→   285→
   286→   286→  const handleProjectImgUpload = async (files: File[]) => {
   287→   287→    if (files.length === 0) return;
   288→   288→    setUploadingProjImg(true);
   289→   289→    try {
   290→   290→      const { urls } = await uploadImages(files);
   291→   291→      setProjectEdit((s) => (s ? { ...s, images: [...s.images, ...urls] } : s));
   292→   292→      toast.success(files.length === 1 ? "Image uploaded" : `${files.length} images uploaded`);
   293→   293→    } catch (e) {
   294→   294→      toast.error("Upload failed", { description: e instanceof Error ? e.message : "" });
   295→   295→    } finally {
   296→   296→      setUploadingProjImg(false);
   297→   297→    }
   298→   298→  };
   299→   299→
   300→   300→  return (
   301→   301→    <div className="space-y-5">
   302→   302→      <div className="flex flex-wrap items-center justify-between gap-3">
   303→   303→        <div>
   304→   304→          <h1 className="font-display text-2xl font-bold text-zinc-900">Clients</h1>
   305→   305→          <p className="mt-1 text-sm text-zinc-500">
   306→   306→            Shown on the homepage — click a logo on the site to view that client&apos;s projects.
   307→   307→          </p>
   308→   308→        </div>
   309→   309→        <Button className="font-bold" onClick={openNewClient}>
   310→   310→          <Plus className="mr-1.5 h-4 w-4" aria-hidden="true" /> Add Client
   311→   311→        </Button>
   312→   312→      </div>
   313→   313→
   314→   314→      {/* Client cards */}
   315→   315→      {loading ? (
   316→   316→        <div className="grid gap-4 sm:grid-cols-2 xl:grid-cols-3">
   317→   317→          {Array.from({ length: 6 }).map((_, i) => (
   318→   318→            <Skeleton key={i} className="h-56 rounded-2xl" />
   319→   319→          ))}
   320→   320→        </div>
   321→   321→      ) : clients.length === 0 ? (
   322→   322→        <div className="rounded-2xl border bg-white p-12 text-center text-sm text-zinc-400 shadow-sm">
   323→   323→          No clients yet — add your first client to show their logo on the homepage.
   324→   324→        </div>
   325→   325→      ) : (
   326→   326→        <div className="grid gap-4 sm:grid-cols-2 xl:grid-cols-3">
   327→   327→          {clients.map((c) => (
   328→   328→            <section key={c.id} className="min-w-0 rounded-2xl border bg-white shadow-sm">
   329→   329→              {/* header: round logo + name */}
   330→   330→              <div className="flex items-center gap-3 border-b p-4">
   331→   331→                <span className="flex h-14 w-14 shrink-0 items-center justify-center overflow-hidden rounded-full border-2 border-zinc-200 bg-white">
   332→   332→                  <img src={c.logo} alt={`${c.name} logo`} className="h-full w-full object-cover" />
   333→   333→                </span>
   334→   334→                <div className="min-w-0 flex-1">
   335→   335→                  <p className="truncate font-display text-base font-bold text-zinc-900">{c.name}</p>
   336→   336→                  <p className="text-xs font-medium uppercase tracking-wider text-zinc-400">
   337→   337→                    {c.industry || "—"}
   338→   338→                  </p>
   339→   339→                </div>
   340→   340→                <div className="flex shrink-0 gap-1.5">
   341→   341→                  <Button variant="ghost" size="icon" className="h-8 w-8" onClick={() => openEditClient(c)} aria-label={`Edit ${c.name}`}>
   342→   342→                    <Pencil className="h-4 w-4" aria-hidden="true" />
   343→   343→                  </Button>
   344→   344→                  <Button variant="ghost" size="icon" className="h-8 w-8 text-red-600 hover:bg-red-50" onClick={() => setDeleteClient(c)} aria-label={`Delete ${c.name}`}>
   345→   345→                    <Trash2 className="h-4 w-4" aria-hidden="true" />
   346→   346→                  </Button>
   347→   347→                </div>
   348→   348→              </div>
   349→   349→
   350→   350→              {/* projects list */}
   351→   351→              <div className="p-4">
   352→   352→                <div className="flex items-center justify-between">
   353→   353→                  <p className="flex items-center gap-1.5 text-xs font-bold uppercase tracking-wider text-zinc-500">
   354→   354→                    <Briefcase className="h-3.5 w-3.5 text-primary" aria-hidden="true" />
   355→   355→                    Projects ({c.projects.length})
   356→   356→                  </p>
   357→   357→                  <button
   358→   358→                    onClick={() => openNewProject(c)}
   359→   359→                    className="flex items-center gap-1 rounded-md border border-primary/40 px-2 py-0.5 text-[11px] font-bold text-primary transition-colors hover:bg-accent"
   360→   360→                  >
   361→   361→                    <Plus className="h-3 w-3" aria-hidden="true" /> Add Project
   362→   362→                  </button>
   363→   363→                </div>
   364→   364→                <ul className="mt-3 max-h-44 space-y-2 overflow-y-auto scrollbar-thin">
   365→   365→                  {c.projects.length === 0 && (
   366→   366→                    <li className="py-3 text-center text-xs text-zinc-400">No projects yet.</li>
   367→   367→                  )}
   368→   368→                  {c.projects.map((p) => (
   369→   369→                    <li
   370→   370→                      key={p.id}
   371→   371→                      className="group flex items-center gap-3 rounded-lg border bg-zinc-50/60 p-2"
   372→   372→                    >
   373→   373→                      <span className="relative shrink-0">
   374→   374→                        <img src={p.images[0]} alt={p.title} className="h-11 w-14 rounded-md border object-cover" />
   375→   375→                        {p.images.length > 1 && (
   376→   376→                          <span className="absolute -right-1.5 -top-1.5 flex items-center gap-0.5 rounded-full bg-zinc-900 px-1.5 py-0.5 text-[9px] font-bold text-white shadow">
   377→   377→                            <Images className="h-2.5 w-2.5" aria-hidden="true" />
   378→   378→                            {p.images.length}
   379→   379→                          </span>
   380→   380→                        )}
   381→   381→                      </span>
   382→   382→                      <div className="min-w-0 flex-1">
   383→   383→                        <p className="truncate text-sm font-semibold text-zinc-800">{p.title}</p>
   384→   384→                        <p className="text-xs text-zinc-400">{p.year ?? "—"}</p>
   385→   385→                      </div>
   386→   386→                      <div className="flex shrink-0 gap-1">
   387→   387→                        <button
   388→   388→                          onClick={() => openEditProject(c, p)}
   389→   389→                          className="flex h-7 w-7 items-center justify-center rounded-md text-zinc-400 hover:bg-white hover:text-primary"
   390→   390→                          aria-label={`Edit ${p.title}`}
   391→   391→                        >
   392→   392→                          <Pencil className="h-3.5 w-3.5" aria-hidden="true" />
   393→   393→                        </button>
   394→   394→                        <button
   395→   395→                          onClick={() => setDeleteProject(p)}
   396→   396→                          className="flex h-7 w-7 items-center justify-center rounded-md text-zinc-400 hover:bg-red-50 hover:text-red-600"
   397→   397→                          aria-label={`Delete ${p.title}`}
   398→   398→                        >
   399→   399→                          <Trash2 className="h-3.5 w-3.5" aria-hidden="true" />
   400→   400→                        </button>
   401→   401→                      </div>
   402→   402→                    </li>
   403→   403→                  ))}
   404→   404→                </ul>
   405→   405→              </div>
   406→   406→            </section>
   407→   407→          ))}
   408→   408→        </div>
   409→   409→      )}
   410→   410→
   411→   411→      {/* ---------- Client dialog ---------- */}
   412→   412→      <Dialog open={clientOpen} onOpenChange={(o) => !o && setClientOpen(false)}>
   413→   413→        <DialogContent className="sm:max-w-2xl">
   414→   414→          {clientEdit && (
   415→   415→            <>
   416→   416→              <DialogHeader>
   417→   417→                <DialogTitle>{clientEdit.id ? "Edit Client" : "Add Client"}</DialogTitle>
   418→   418→                <DialogDescription>The logo appears in a round frame on the homepage.</DialogDescription>
   419→   419→              </DialogHeader>
   420→   420→              <div className="space-y-4">
   421→   421→                {/* logo preview + upload */}
   422→   422→                <div className="flex items-center gap-4">
   423→   423→                  <span className="flex h-20 w-20 shrink-0 items-center justify-center overflow-hidden rounded-full border-2 border-zinc-200 bg-white">
   424→   424→                    {clientEdit.logo ? (
   425→   425→                      <img src={clientEdit.logo} alt="Client logo preview" className="h-full w-full object-cover" />
   426→   426→                    ) : (
   427→   427→                      <span className="text-[10px] font-semibold text-zinc-300">LOGO</span>
   428→   428→                    )}
   429→   429→                  </span>
   430→   430→                  <label className="flex h-10 cursor-pointer items-center gap-2 rounded-md border border-dashed px-4 text-sm font-semibold text-zinc-500 hover:border-primary/50 hover:text-primary">
   431→   431→                    {uploadingLogo ? <Loader2 className="h-4 w-4 animate-spin" aria-hidden="true" /> : <Upload className="h-4 w-4" aria-hidden="true" />}
   432→   432→                    Upload logo
   433→   433→                    <input
   434→   434→                      type="file"
   435→   435→                      accept="image/png,image/jpeg,image/webp"
   436→   436→                      className="hidden"
   437→   437→                      onChange={(e) => {
   438→   438→                        const f = e.target.files?.[0];
   439→   439→                        if (f) void handleLogoUpload(f);
   440→   440→                        e.currentTarget.value = "";
   441→   441→                      }}
   442→   442→                    />
   443→   443→                  </label>
   444→   444→                </div>
   445→   445→                <div className="space-y-1.5">
